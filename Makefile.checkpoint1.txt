# 🧹 CLEANUP

.PHONY: clean-venv
clean-venv:
	@echo Removing virtual environment...

	@$(SYSTEM_PYTHON) -c "import shutil, pathlib; \
venv = pathlib.Path('$(VENV_NAME)'); \
shutil.rmtree(venv, ignore_errors=True) if venv.exists() else None"

	@echo Virtual environment removed.

.PHONY: docker-remove
docker-remove:
	@echo Removing Docker images...

	-$(DOCKER) rmi $(DOCKER_IMAGE_BASE)
	-$(DOCKER) rmi $(DOCKER_IMAGE_DEV)
	-$(DOCKER) rmi $(DOCKER_IMAGE_PROD)

	@echo Docker image cleanup completed.

.PHONY: prune
prune:
	@echo Running Docker system prune...
	$(DOCKER) system prune -f
	@echo Docker prune completed.

.PHONY: prune-all
prune-all:
	$(DOCKER) system prune -a -f --volumes

# Linux shell
.PHONY: clean-cache-linux
clean-cache-linux:
	rm -rf .pytest_cache
	rm -rf .ruff_cache
	rm -rf build
	rm -rf dist
	rm -rf *.egg-info

# Universal
.PHONY: clean-cache
clean-cache:
	@echo Cleaning cache files...

	@$(SYSTEM_PYTHON) -c "import shutil, pathlib; \
paths = [ \
'.pytest_cache', \
'.ruff_cache', \
'build', \
'dist' \
]; \
[shutil.rmtree(p, ignore_errors=True) for p in paths if pathlib.Path(p).exists()]"

	@$(SYSTEM_PYTHON) -c "import pathlib, shutil; \
[shutil.rmtree(p, ignore_errors=True) for p in pathlib.Path('.').glob('*.egg-info')]"

	@echo Cache cleanup completed.

# Linux shell
.PHONY: clean-pyc
clean-pyc:
	find . -type f -name "*.pyc" -delete
	find . -type d -name "__pycache__" -delete

# Universal
.PHONY: clean-pyc
clean-pyc:
	@echo Cleaning Python cache files...

	@$(SYSTEM_PYTHON) -c "import pathlib; \
[p.unlink() for p in pathlib.Path('.').rglob('*.pyc')]"

	@$(SYSTEM_PYTHON) -c "import pathlib, shutil; \
[shutil.rmtree(p, ignore_errors=True) for p in pathlib.Path('.').rglob('__pycache__')]"

	@echo Python cache cleanup completed.

.PHONY: clean-all
clean-all: clean-cache clean-pyc
	@echo Full cleanup completed.
